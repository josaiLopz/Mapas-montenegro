		function filtros_base(){
			$cond=array();
		$cond[]=array('Base.lat'=>'<>0');
		$cond[]=array('Base.lng'=>'<>0');

		if($this->params['form']['territorios']=='true'){
			$cond[]=array('Base.id_distribuidor'=>'>0');
		}

		if(!empty($this->params['form'])){
			if(!empty($this->params['form']['nombre']))
				$cond[]='Base.nombre like "%'.$this->params['form']['nombre'].'%"';

			if(!empty($this->params['form']['estado']))
				$cond[]=array('Base.estado'=>$this->params['form']['estado']);
			if(!empty($this->params['form']['municipio']) && $this->params['form']['municipio']!='Cualquiera')
				$cond[]=array('Base.municipio'=>$this->params['form']['municipio']);
			if(!empty($this->params['form']['sector']) && $this->params['form']['sector']!='Cualquiera')
				$cond[]='Base.sector like "%'.$this->params['form']['sector'].'%"';

			if(!empty($this->params['form']['tipo']) && $this->params['form']['tipo']!='Cualquiera')
				$cond[]=array('Base.tipo'=>$this->params['form']['tipo']);
		
			if(!empty($this->params['form']['turno']) && $this->params['form']['turno']!='Cualquiera')
				$cond[]=array('Base.turno'=>$this->params['form']['turno']);

			
			if(!empty($this->params['form']['distribuidor'])){
				if($this->params['form']['distribuidor']==99999)
					$cond[]=array('Base.id_distribuidor'=>'<1');
				else
					$cond[]=array('Base.id_distribuidor'=>$this->params['form']['distribuidor']);
			}
			if(strlen($this->params['form']['alumnos'])>0){
				$als=$this->params['form']['alumnos'];
				$als=str_replace("-","<",$als);
				$als=str_replace("+",">",$als);
				$cond[]=array('Base.alumnos'=>$als);
			}
			
			if(!empty($this->params['form']['cct']))
				$cond[]='Base.cct like "%'.$this->params['form']['cct'].'%"';

			if(!empty($this->params['form']['estatus']) && $this->params['form']['estatus']!='Cualquiera'){
			
				if($this->params['form']['estatus']==1)
				$cond[]=array('Base.estatus'=>array(0,1));
				
				else
				$cond[]=array('Base.estatus'=>$this->params['form']['estatus']);
			}

				
			if(!empty($this->params['form']['verificada']) && $this->params['form']['verificada']!='Cualquiera'){
				if($this->params['form']['verificada']==1)
					$cond[]=array('Base.verificada'=>array(0,1));
				
				else
					$cond[]=array('Base.verificada'=>$this->params['form']['verificada']);

			}

		}
		return $cond;
		}

	function guardar() {
		// 1. Preparación inicial de datos (sin cambios)
		$datos = array();
		$datos['Base'] = $this->params['form'];
		$datos['Base']['id'] = trim($datos['Base']['id']);
		
		// Procesamiento común de datos (sin cambios)
		if(empty($datos['Base']['alumnos'])) {
			$datos['Base']['alumnos'] = 0;
		}
	
		if($datos['Base']['sector'] == 'blico') {
			$datos['Base']['sector'] = utf8_decode('PÚBLICO');
		}
	
		// Obtención de datos de estado (sin cambios)
		$estado = $this->Estado->find(array(
			'Estado.estado' => $datos['Base']['estado'],
			'Estado.municipio' => $datos['Base']['municipio']
		));
		
		if(!empty($estado)) {
			$datos['Base']['estado_n'] = $estado['Estado']['estado_n'];
			$datos['Base']['municipio_n'] = $estado['Estado']['municipio_n'];
		}
	
		// ==============================================
		// CAMBIO PRINCIPAL: Lógica diferencial para nuevo vs edición
		// ==============================================
		$mover = 0; // Flag para nuevos registros
		
		if(empty($datos['Base']['id'])) {
			// ==========================================
			// CASO 1: NUEVO REGISTRO
			// ==========================================
			$datos['Base'] = array_reverse($datos['Base']);
			array_pop($datos['Base']);
			$this->Base->create();
			
			// Asignación de valores por defecto
			$datos['Base']['lat'] = 23.6;
			$datos['Base']['lng'] = -102.5;
			$datos['Base']['nombre'] = utf8_decode($datos['Base']['nombre']);
			
			// Guardado PRINCIPAL para nuevo registro
			$this->Base->save($datos);
			
			$mover = 1; // Activar flag de movimiento para nuevos registros
		} else {
			// ==========================================
			// CASO 2: EDICIÓN DE REGISTRO EXISTENTE
			// ==========================================
			// Solo guardamos en el historial para ediciones
			$this->guardarHistorial($datos['Base']);
			
			// Convertimos el nombre a formato correcto
			$datos['Base']['nombre'] = utf8_decode($datos['Base']['nombre']);
		}
	
		// ==============================================
		// Resto de la lógica original (sin cambios)
		// ==============================================
		$mis_permisos = $this->Session->read('Usuario.permisos');
		$permiso_editar = 0;
		$c_permisos = explode("@", $mis_permisos);
		if(array_search("Bases|add", $c_permisos) !== false) {
			$permiso_editar = 1;
		}
	
		configure::load("app/app"); 
		$escuelas = configure::read("escuelas"); 
		$distribuidores = $this->Escuela->Usuario->distribuidores();
	
		$cad = "<div style='width:200px;padding:10px;'>";
	
		if($datos['Base']['verificada'] == 2) {
			$cad .= '<div style="position:absolute;left:5px;top:5px"><img src="/img/iconos/calificaciones.png"></div>';
		}
	
		$cad .= "<span class='titulo_etiqueta'>Nombre:</span> ".utf8_encode($datos['Base']['nombre']);
		$cad .= "<br><span class='titulo_etiqueta'>Estado:</span> ".utf8_encode($datos['Base']['estado_n']);
		$cad .= "<br><span class='titulo_etiqueta'>Municipio:</span> ".utf8_encode($datos['Base']['municipio_n']);
		$cad .= "<br><span class='titulo_etiqueta'>Tipo:</span> ".$datos['Base']['tipo'];
		$cad .= "<br><span class='titulo_etiqueta'>Sector:</span> ".utf8_encode($datos['Base']['sector']);
		$cad .= "<br><span class='titulo_etiqueta'>Turno:</span> ".$escuelas['turnos'][$datos['Base']['turno']];
		$cad .= "<br><span class='titulo_etiqueta'>CCT:</span> ".$datos['Base']['cct'];
		$cad .= "<br><span class='titulo_etiqueta'># De Alumnos:</span> ".$datos['Base']['alumnos'];
		$cad .= "<br><span class='titulo_etiqueta'>Distribuidor:</span> ".$distribuidores[$datos['Base']['id_distribuidor']];
	
		if($permiso_editar == 1) {
			$cad .= "<br><br><center><a href=javascript:editar_info(".$datos['Base']['id'].")>Editar información</a></center>";
			$cad .= "<br><center><a href=javascript:mover_pin(".$datos['Base']['id'].")>Mover Pin</a></center>";
		}
	
		$cad .= "</div>";
		echo $cad;
	
		// Lógica de movimiento para nuevos registros (sin cambios)
		if($mover) {
			?>
			<script>
				document.getElementById("BaseCct").value='<?php echo $datos['Base']['cct']; ?>';
				setTimeout('mover_pin(<?php echo $this->Base->id; ?>)',1000);
			</script>
			<?php
		}
		die;
	}
	function guardarHistorial($datos) {
		// Configuración de la conexión a la base de datos
		$db_host = 'localhost'; // Cambia esto según tu configuración
		$db_name = 'prueba'; // Nombre de tu base de datos
		$db_user = 'root'; // Usuario de la base de datos
		$db_pass = 'root'; // Contraseña de la base de datos
		
		try {
			// Conexión a la base de datos con PDO
			$conn = new PDO("mysql:host=$db_host;dbname=$db_name", $db_user, $db_pass);
			$conn->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
			
			$sql_check = "SELECT COUNT(*) as count FROM bases_historial WHERE id_original = :id";
			$stmt_check = $conn->prepare($sql_check);
			$stmt_check->bindParam(':id', $datos['id']);
			$stmt_check->execute();
			$result = $stmt_check->fetch(PDO::FETCH_ASSOC);
			
			// Si ya existe un registro, no hacemos la inserción
			if ($result['count'] > 0) {
				echo "<script>alert('Ya existe una edición pendiente. Por favor espere...');</script>";
				return false; // O puedes lanzar una excepción o registrar un mensaje
			}
			

			// Preparamos los datos para la inserción
			$datos['nombre'] = utf8_decode($datos['nombre']);
			$datos['fecha_modificacion'] = date('Y-m-d H:i:s');
			$datos['usuario_modificacion'] = $this->Session->read('Usuario.id');
			
			// Consulta SQL para insertar en la tabla de historial
			$sql = "INSERT INTO bases_historial (
				id_original, cct, nombre, turno, tipo, sector, 
				estado, estado_n, municipio, municipio_n, 
				alumnos, lat, lng, id_distribuidor, grupos, nombre_contacto, telefono_contacto, correo_contacto, notas,
				verificada, estatus, fecha_modificacion, usuario_modificacion
			) VALUES (
				:id, :cct, :nombre, :turno, :tipo, :sector, 
				:estado, :estado_n, :municipio, :municipio_n, 
				:alumnos, :lat, :lng, :id_distribuidor, :grupos, :nombre_contacto, :telefono_contacto, :correo_contacto, :notas,
				:verificada, :estatus, :fecha_modificacion, :usuario_modificacion
			)";
			
			$stmt = $conn->prepare($sql);
			
			// Bind de parámetros

	public function contar_registros_transferencia_ajax() {
        $this->layout = 'ajax'; // No usa layout completo
        $this->autoRender = false; // No renderiza un archivo .ctp

        // Para CakePHP 1.x, los parámetros GET están en $this->params['url']
        $id_usuario_origen = isset($this->params['url']['usuario_origen']) ? intval($this->params['url']['usuario_origen']) : null;
        $tipo = isset($this->params['url']['tipo']) ? $this->params['url']['tipo'] : null;
        $turno = isset($this->params['url']['turno']) ? $this->params['url']['turno'] : null; // Será '1', '2', etc.
        $sector = isset($this->params['url']['sector']) ? $this->params['url']['sector'] : null;

        if (empty($id_usuario_origen)) {
            echo json_encode(array(
                'success' => false,
                'message' => 'El Usuario Origen es obligatorio para el conteo.',
                'count' => 0,
                'total_alumnado' => 0 
            ));
            return;
        }

        $db = $this->Base->getDataSource();
        $conditions = array();
        $conditions[] = "id_distribuidor = " . $db->value($id_usuario_origen, 'integer');

        if (!empty($tipo)) {
            $conditions[] = "tipo = " . $db->value($tipo, 'string');
        }
        if (!empty($turno)) {
            $conditions[] = "turno = " . $db->value($turno, 'string');
        }
        if (!empty($sector)) {
            $conditions[] = "sector = " . $db->value($sector, 'string');
        }
        
        $where_clause = implode(" AND ", $conditions);
        $count = 0;
        $total_alumnado = 0;

        // La consulta se realiza sobre la tabla 'bases'
        // COALESCE(SUM(alumnos), 0) asegura que si no hay registros o alumnos es NULL, devuelva 0.
        $query_sql = "SELECT COUNT(*) as total_registros, COALESCE(SUM(alumnos), 0) as total_alumnado FROM bases WHERE " . $where_clause;
        $result_array = $this->Base->query($query_sql);
        
        if (!empty($result_array) && isset($result_array[0][0])) {
            $count = (int)$result_array[0][0]['total_registros'];
            $total_alumnado = (int)$result_array[0][0]['total_alumnado'];
        }
        
        echo json_encode(array(
            'success' => true,
            'count' => $count,
            'total_alumnado' => $total_alumnado 
        ));
    }

			'PRIVADO' => 'Privado'
		);

			$this->autoRender = false;

			$id_usuario_destino = isset($this->data['Base']['usuario_destino']) ? intval($this->data['Base']['usuario_destino']) : null;
			
			$tipo_filtro = isset($this->data['Base']['tipo_filtro']) ? $this->data['Base']['tipo_filtro'] : null;
			$turno_filtro = isset($this->data['Base']['turno_filtro']) ? $this->data['Base']['turno_filtro'] : null;
			$sector_filtro = isset($this->data['Base']['sector_filtro']) ? $this->data['Base']['sector_filtro'] : null;

			if (empty($id_usuario_origen) || empty($id_usuario_destino)) {
				echo json_encode(array(
					'success' => false,
					'message' => 'Debe seleccionar un usuario origen y un usuario destino.'
				));
				return;
			}
			
			if ($id_usuario_origen == $id_usuario_destino) {
				echo json_encode(array(
					'success' => false,
					'message' => 'El usuario origen y destino no pueden ser el mismo.'
				));
				return;
			}

			$dataSource = ConnectionManager::getDataSource('default'); 
			$db = $this->Base->getDataSource(); 

			$conditions_transfer = array();
			$conditions_transfer[] = "id_distribuidor = " . $db->value($id_usuario_origen, 'integer');

			if (!empty($tipo_filtro)) {
				$conditions_transfer[] = "tipo = " . $db->value($tipo_filtro, 'string');
			}
			if (!empty($turno_filtro)) {
				$conditions_transfer[] = "turno = " . $db->value($turno_filtro, 'string');
			}
			if (!empty($sector_filtro)) {
				$conditions_transfer[] = "sector = " . $db->value($sector_filtro, 'string');
			}
			$where_clause_transfer = implode(" AND ", $conditions_transfer);
			
			try {
				$dataSource->begin(); 

				$update_bases_sql = "UPDATE bases SET id_distribuidor = " . $db->value($id_usuario_destino, 'integer') . " WHERE " . $where_clause_transfer;
				$this->Base->query($update_bases_sql);
				$affected_bases = $this->Base->getAffectedRows();

				$update_escuelas_sql = "UPDATE escuelas SET id_distribuidor = " . $db->value($id_usuario_destino, 'integer') . " WHERE " . $where_clause_transfer;
				$this->Escuela->query($update_escuelas_sql);
				$affected_escuelas = $this->Escuela->getAffectedRows();

				$this->Escuela->Usuario->query("
					UPDATE usuarios 
					SET activo = 1 
					WHERE id = " . $db->value($id_usuario_origen, 'integer')
				);
				
				$dataSource->commit(); 
				echo json_encode(array(
					'success' => true,
					'message' => "Transferencia completada. Registros afectados en Bases: $affected_bases, en Escuelas: $affected_escuelas."
				));
			} catch (Exception $e) {
				$dataSource->rollback(); 
				echo json_encode(array(
					'success' => false,
					'message' => 'Error al transferir datos: ' . $e->getMessage()
				));
			}
			return; 
		} 
		
		$this->set(compact('usuariosOptions', 'tiposOptions', 'turnosOptions', 'sectoresOptions'));
		$this->render('transferir_usuarios'); 
	}

	public function contar_registros_ajax() {
        $this->layout = 'ajax'; // No usa layout completo
        $this->autoRender = false; // No renderiza un archivo .ctp

        // Para CakePHP 1.x, los parámetros GET están en $this->params['url']
        $estado_origen = isset($this->params['url']['estado_origen']) ? intval($this->params['url']['estado_origen']) : null;
        $tipo = isset($this->params['url']['tipo']) ? $this->params['url']['tipo'] : null;
        $turno = isset($this->params['url']['turno']) ? $this->params['url']['turno'] : null;
        $sector = isset($this->params['url']['sector']) ? $this->params['url']['sector'] : null;

        if (empty($estado_origen)) {
            echo json_encode(array(
                'success' => false,
                'message' => 'El estado es obligatorio.',
                'count' => 0,
                'total_alumnado' => 0 // Aseguramos que siempre exista este campo en la respuesta
            ));
            return;
        }

        $db = $this->Base->getDataSource();
        $conditions = array();
        $conditions[] = "estado = " . $db->value($estado_origen, 'integer');

        if (!empty($tipo)) {
            $conditions[] = "tipo = " . $db->value($tipo, 'string');
        }
        if (!empty($turno)) {
            // La columna 'turno' en la BD debe ser compatible con '1', '2', etc.
            $conditions[] = "turno = " . $db->value($turno, 'string');
        }
        if (!empty($sector)) {
            $conditions[] = "sector = " . $db->value($sector, 'string');
        }
        
        $where_clause = implode(" AND ", $conditions);
        $count = 0;
        $total_alumnado = 0; // Inicializar el total de alumnado

        if (!empty($where_clause)) {
            // Modificamos la consulta para obtener COUNT(*) y SUM(alumnos)
            // Usamos COALESCE para asegurar que SUM(alumnos) devuelva 0 si no hay registros o todos los alumnos son NULL
            $query_sql = "SELECT COUNT(*) as total_registros, COALESCE(SUM(alumnos), 0) as total_alumnado FROM bases WHERE " . $where_clause;
            $result_array = $this->Base->query($query_sql);
            
            if (!empty($result_array) && isset($result_array[0][0])) {
                $count = (int)$result_array[0][0]['total_registros'];
                $total_alumnado = (int)$result_array[0][0]['total_alumnado']; // Obtenemos la suma del alumnado
            }
        }
        
        // Devolvemos ambos datos en la respuesta JSON
        echo json_encode(array(
            'success' => true,
            'count' => $count,
            'total_alumnado' => $total_alumnado 
        ));
    }

	function asignar_distribuidor_estado() {
		$this->layout = false;
		$this->autoRender = false;

		if (!empty($this->data)) { // Handles POST request
			$estado_origen = intval($this->data['Base']['estado_origen']);

			$tipo = $this->data['Base']['tipo']; // New field
			$turno = $this->data['Base']['turno']; // New field
			$sector = $this->data['Base']['sector']; // New field
			
			$db = $this->Base->getDataSource();
			$conditions = array();
			// Always filter by estado
			$conditions[] = "estado = " . $db->value($estado_origen, 'integer');

			// Add conditions for new filters if a specific option is selected (not "Todos")
			if (!empty($tipo)) {
				$conditions[] = "tipo = " . $db->value($tipo, 'string');
			}
			if (!empty($turno)) {
				$conditions[] = "turno = " . $db->value($turno, 'string');
			}
			if (!empty($sector)) {
				// Assuming DB stores 'PÚBLICO' with accent based on other controller actions
				$conditions[] = "sector = " . $db->value($sector, 'string');
			}
			
			$where_clause = implode(" AND ", $conditions);
			$query = "UPDATE bases SET id_distribuidor = " . $db->value($id_usuario_nuevo, 'integer') . " WHERE " . $where_clause;

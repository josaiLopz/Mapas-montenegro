		function filtros_base(){
			$cond=array();
		$cond[]=array('Base.lat'=>'<>0');
		$cond[]=array('Base.lng'=>'<>0');

		if($this->params['form']['territorios']=='true'){
			$cond[]=array('Base.id_distribuidor'=>'>0');
		}

		if(!empty($this->params['form'])){
			if(!empty($this->params['form']['nombre']))
				$cond[]='Base.nombre like "%'.$this->params['form']['nombre'].'%"';

			if(!empty($this->params['form']['estado']))
				$cond[]=array('Base.estado'=>$this->params['form']['estado']);
			if(!empty($this->params['form']['municipio']) && $this->params['form']['municipio']!='Cualquiera')
				$cond[]=array('Base.municipio'=>$this->params['form']['municipio']);
			if(!empty($this->params['form']['sector']) && $this->params['form']['sector']!='Cualquiera')
				$cond[]='Base.sector like "%'.$this->params['form']['sector'].'%"';

			if(!empty($this->params['form']['tipo']) && $this->params['form']['tipo']!='Cualquiera')
				$cond[]=array('Base.tipo'=>$this->params['form']['tipo']);
		
			if(!empty($this->params['form']['turno']) && $this->params['form']['turno']!='Cualquiera')
				$cond[]=array('Base.turno'=>$this->params['form']['turno']);

			
			if(!empty($this->params['form']['distribuidor'])){
				if($this->params['form']['distribuidor']==99999)
					$cond[]=array('Base.id_distribuidor'=>'<1');
				else
					$cond[]=array('Base.id_distribuidor'=>$this->params['form']['distribuidor']);
			}
			if(strlen($this->params['form']['alumnos'])>0){
				$als=$this->params['form']['alumnos'];
				$als=str_replace("-","<",$als);
				$als=str_replace("+",">",$als);
				$cond[]=array('Base.alumnos'=>$als);
			}
			
			if(!empty($this->params['form']['cct']))
				$cond[]='Base.cct like "%'.$this->params['form']['cct'].'%"';

			if(!empty($this->params['form']['estatus']) && $this->params['form']['estatus']!='Cualquiera'){
			
				if($this->params['form']['estatus']==1)
				$cond[]=array('Base.estatus'=>array(0,1));
				
				else
				$cond[]=array('Base.estatus'=>$this->params['form']['estatus']);
			}

				
			if(!empty($this->params['form']['verificada']) && $this->params['form']['verificada']!='Cualquiera'){
				if($this->params['form']['verificada']==1)
					$cond[]=array('Base.verificada'=>array(0,1));
				
				else
					$cond[]=array('Base.verificada'=>$this->params['form']['verificada']);

			}

		}
		return $cond;
		}

	function transferir_usuarios() {
		$this->layout = false;
		$this->autoRender = false; // Para devolver JSON
		
		// Obtener usuarios activos
		$usuarios_raw = $this->Escuela->Usuario->find('all', array(
			'conditions' => array('Usuario.activo' => 1),
			'fields' => array('Usuario.id', 'Usuario.nombre','Usuario.apellido_p','Usuario.apellido_m'),
			'order' => array('Usuario.nombre' => 'ASC')
		));
		
		$usuarios = array();
		foreach ($usuarios_raw as $u) {
			$usuarios[$u['Usuario']['id']] = trim(
				$u['Usuario']['nombre'] . ' ' . $u['Usuario']['apellido_p'] . ' ' . $u['Usuario']['apellido_m']
			);
		}
		
		if (!empty($this->data)) {
			$id_usuario_viejo = $this->data['Base']['usuario_origen'];
			$id_usuario_nuevo = $this->data['Base']['usuario_destino'];
			
			if ($id_usuario_viejo == $id_usuario_nuevo) {
				echo json_encode(array(
					'success' => false,
					'message' => 'El usuario origen y destino no pueden ser el mismo.'
				));
				return;
			}
			
			// Obtener el datasource para la transacción
			$dataSource = ConnectionManager::getDataSource('default');
			
			try {
				// Iniciar transacción
				$dataSource->begin();
				
				// 1. Transferir BASES
				$this->Base->query("
					UPDATE bases 
					SET id_distribuidor = $id_usuario_nuevo 
					WHERE id_distribuidor = $id_usuario_viejo
				");
				
				// 2. Transferir ESCUELAS
				$this->Escuela->query("
					UPDATE escuelas 
					SET id_distribuidor = $id_usuario_nuevo 
					WHERE id_distribuidor = $id_usuario_viejo
				");
				
				// 3. Desactivar usuario viejo
				$this->Escuela->Usuario->query("
					UPDATE usuarios 
					SET activo = 1 
					WHERE id = $id_usuario_viejo
				");
				
				$dataSource->commit();
				echo json_encode(array(
					'success' => true,
					'message' => 'Transferencia completada correctamente.'
				));
			} catch (Exception $e) {
				$dataSource->rollback();
				echo json_encode(array(
					'success' => false,
					'message' => 'Error al transferir datos: ' . $e->getMessage()
				));
			}
		} else {
			// Si no es POST, mostrar el formulario
			$this->set('usuarios', $usuarios);
			$this->render('transferir_usuarios');
		}
	}
	
	function asignar_distribuidor_estado() {
		$this->layout = false;
		$this->autoRender = false; // Para devolver JSON
		
		// Procesar el formulario si se envió
		if (!empty($this->data)) {
			$estado_origen = intval($this->data['Base']['estado_origen']);
			$id_usuario_nuevo = intval($this->data['Base']['id_usuario_nuevo']);
			
			$query = "UPDATE bases SET id_distribuidor = {$id_usuario_nuevo} WHERE estado = {$estado_origen}";
			$result = $this->Base->query($query);
			
			if ($result !== false) {
				$filas_afectadas = $this->Base->getAffectedRows();
				echo json_encode(array(
					'success' => true,
					'message' => "✔ Distribuidor asignado correctamente a $filas_afectadas registros."
				));
			} else {
				$error = $this->Base->getDataSource()->lastError();
				echo json_encode(array(
					'success' => false,
					'message' => "❌ Error al asignar distribuidor: " . $error
				));
			}
			return;
		}
		
		// Obtener estados
		$estados = $this->Estado->query("
			SELECT estado, estado_n 
			FROM estados 
			GROUP BY estado_n 
			ORDER BY estado_n
		");
		
		$estadosOptions = array('' => '-- Selecciona un estado --');
		foreach ($estados as $estado) {
			$estadosOptions[$estado['estados']['estado']] = $estado['estados']['estado_n'];
		}
		
		// Obtener usuarios
		$usuarios = $this->Escuela->Usuario->query("
			SELECT id, nombre, apellido_p, apellido_m 
			FROM usuarios 
			ORDER BY nombre
		");
		
		$usuariosOptions = array('' => '-- Selecciona un usuario --');
		foreach ($usuarios as $usuario) {
			$nombre_completo = trim(
				$usuario['usuarios']['nombre'] . ' ' .
				$usuario['usuarios']['apellido_p'] . ' ' .
				$usuario['usuarios']['apellido_m']
			);
			$usuariosOptions[$usuario['usuarios']['id']] = $nombre_completo;
		}
	
		$this->set(compact('estadosOptions', 'usuariosOptions'));
		$this->render('asignar_distribuidor_estado');
	}
